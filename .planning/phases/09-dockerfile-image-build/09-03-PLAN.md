    Create `/home/rachel/rachel8/scripts/test-container.sh` as an executable integration test script. This script builds the image, launches a test container, and verifies all Phase 9 requirements automatically. It uses a dedicated test user ID to avoid conflicts with production containers.

    **Shebang and strict mode:**
    ```bash
    #!/usr/bin/env bash
    set -euo pipefail
    ```

    **Required env vars (must be set before running):**
    - `TELEGRAM_BOT_TOKEN` - A real Telegram bot token for testing
    - `OWNER_TELEGRAM_USER_ID` - The tester's Telegram user ID

    **Test configuration:**
    ```bash
    TEST_USER_ID="test-integration"
    CONTAINER_NAME="rachel-user-${TEST_USER_ID}"
    VOLUME_NAME="rachel-user-${TEST_USER_ID}-data"
    SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
    REPO_ROOT="$(cd "${SCRIPT_DIR}/.." && pwd)"
    PASS=0
    FAIL=0
    ```

    **Helper functions:**
    ```bash
    pass() { echo "  PASS: $1"; PASS=$((PASS + 1)); }
    fail() { echo "  FAIL: $1"; FAIL=$((FAIL + 1)); }
    check() {
      if eval "$2"; then
        pass "$1"
      else
        fail "$1"
      fi
    }
    ```

    **Cleanup function (always runs, even on failure):**
    ```bash
    cleanup() {
      echo ""
      echo "Cleaning up test container..."
      docker stop "${CONTAINER_NAME}" 2>/dev/null || true
      docker rm "${CONTAINER_NAME}" 2>/dev/null || true
      # Keep volume for inspection if tests failed
      if [[ "$FAIL" -eq 0 ]]; then
        docker volume rm "${VOLUME_NAME}" 2>/dev/null || true
      fi
    }
    trap cleanup EXIT
    ```

    **Test sequence:**

    ```bash
    echo "============================================"
    echo "Rachel8 Docker Integration Test"
    echo "============================================"
    echo ""

    # --- Step 1: Build image ---
    echo "[Step 1/7] Building Docker image..."
    bash "${SCRIPT_DIR}/build.sh"
    echo ""

    check "Image rachel8:latest exists" \
      "docker image inspect rachel8:latest &>/dev/null"

    IMAGE_SIZE=$(docker images rachel8:latest --format '{{.Size}}')
    echo "  Image size: ${IMAGE_SIZE}"
    echo ""

    # --- Step 2: Launch test container ---
    echo "[Step 2/7] Launching test container..."
    export USER_ID="${TEST_USER_ID}"
    # TELEGRAM_BOT_TOKEN and OWNER_TELEGRAM_USER_ID already exported by caller
    bash "${SCRIPT_DIR}/run-container.sh"
    echo ""

    # --- Step 3: Wait for startup ---
    echo "[Step 3/7] Waiting for container to start (max 30s)..."
    STARTED=false
    for i in $(seq 1 30); do
      if docker logs "${CONTAINER_NAME}" 2>&1 | grep -q "Rachel8 is running"; then
        STARTED=true
        echo "  Container ready after ${i}s"
        break
      fi
      sleep 1
    done

    if [[ "$STARTED" == "true" ]]; then
      pass "Rachel8 started within 30 seconds"
    else
      fail "Rachel8 did not start within 30 seconds"
      echo "  Last logs:"
      docker logs --tail 20 "${CONTAINER_NAME}" 2>&1 | sed 's/^/    /'
    fi
    echo ""

    # --- Step 4: Verify container config ---
    echo "[Step 4/7] Verifying container configuration..."

    # Check non-root user
    CONTAINER_USER=$(docker exec "${CONTAINER_NAME}" whoami 2>/dev/null || echo "unknown")
    check "Runs as non-root user" \
      "[[ '${CONTAINER_USER}' != 'root' ]]"

    CONTAINER_UID=$(docker exec "${CONTAINER_NAME}" id -u 2>/dev/null || echo "unknown")
    check "UID is 1000" \
      "[[ '${CONTAINER_UID}' == '1000' ]]"

    # Check memory limit
    MEM_LIMIT=$(docker inspect "${CONTAINER_NAME}" --format '{{.HostConfig.Memory}}')
    # 512MB = 536870912 bytes
    check "Memory limit is 512MB" \
      "[[ '${MEM_LIMIT}' == '536870912' ]]"

    # Check CPU limit
    CPU_NANO=$(docker inspect "${CONTAINER_NAME}" --format '{{.HostConfig.NanoCpus}}')
    # 0.5 CPU = 500000000 nanocpus
    check "CPU limit is 0.5 cores" \
      "[[ '${CPU_NANO}' == '500000000' ]]"

    # Check restart policy
    RESTART=$(docker inspect "${CONTAINER_NAME}" --format '{{.HostConfig.RestartPolicy.Name}}')
    check "Restart policy is unless-stopped" \
      "[[ '${RESTART}' == 'unless-stopped' ]]"

    # Check read-only filesystem
    READ_ONLY=$(docker inspect "${CONTAINER_NAME}" --format '{{.HostConfig.ReadonlyRootfs}}')
    check "Root filesystem is read-only" \
      "[[ '${READ_ONLY}' == 'true' ]]"

    # Check cap-drop ALL
    CAPS=$(docker inspect "${CONTAINER_NAME}" --format '{{.HostConfig.CapDrop}}')
    check "All capabilities dropped" \
      "[[ '${CAPS}' == *'ALL'* ]]"

    # Check no-new-privileges
    SEC_OPTS=$(docker inspect "${CONTAINER_NAME}" --format '{{.HostConfig.SecurityOpt}}')
    check "no-new-privileges enabled" \
      "[[ '${SEC_OPTS}' == *'no-new-privileges'* ]]"

    # Check volume mount
    MOUNTS=$(docker inspect "${CONTAINER_NAME}" --format '{{range .Mounts}}{{.Destination}}{{end}}')
    check "/data volume mounted" \
      "[[ '${MOUNTS}' == *'/data'* ]]"

    # Check .env stub exists
    ENV_EXISTS=$(docker exec "${CONTAINER_NAME}" test -f /app/.env && echo "yes" || echo "no")
    check ".env stub exists at /app/.env" \
      "[[ '${ENV_EXISTS}' == 'yes' ]]"

    echo ""

    # --- Step 5: Verify data persistence ---
    echo "[Step 5/7] Verifying data persistence..."

    # Write a test file to /data
    docker exec "${CONTAINER_NAME}" sh -c "echo 'persistence-test' > /data/test-file.txt" 2>/dev/null
    check "Can write to /data volume" \
      "docker exec '${CONTAINER_NAME}' cat /data/test-file.txt 2>/dev/null | grep -q 'persistence-test'"

    # Verify read-only root filesystem (write should fail)
    WRITE_FAILED=$(docker exec "${CONTAINER_NAME}" sh -c "touch /app/should-fail 2>&1" && echo "no" || echo "yes")
    check "Cannot write to /app (read-only root)" \
      "[[ '${WRITE_FAILED}' == 'yes' ]]"

    echo ""

    # --- Step 6: Verify restart policy ---
    echo "[Step 6/7] Verifying auto-restart on crash..."

    # Kill the main process -- container should auto-restart
    docker exec "${CONTAINER_NAME}" sh -c "kill 1" 2>/dev/null || true
    sleep 3

    RUNNING=$(docker inspect "${CONTAINER_NAME}" --format '{{.State.Running}}' 2>/dev/null || echo "false")
    # Container may still be restarting, check RestartCount
    RESTART_COUNT=$(docker inspect "${CONTAINER_NAME}" --format '{{.RestartCount}}' 2>/dev/null || echo "0")
    check "Container restarted after crash (restart count > 0 or still running)" \
      "[[ '${RUNNING}' == 'true' || '${RESTART_COUNT}' -gt 0 ]]"

    # Wait for restart to complete
    sleep 5
    RUNNING_AFTER=$(docker inspect "${CONTAINER_NAME}" --format '{{.State.Running}}' 2>/dev/null || echo "false")
    check "Container is running after restart" \
      "[[ '${RUNNING_AFTER}' == 'true' ]]"

    echo ""

    # --- Step 7: Summary ---
    echo "============================================"
    echo "Test Results: ${PASS} passed, ${FAIL} failed"
    echo "============================================"

    if [[ "$FAIL" -gt 0 ]]; then
      echo ""
      echo "Some tests failed. Container logs:"
      docker logs --tail 30 "${CONTAINER_NAME}" 2>&1 | sed 's/^/  /'
      exit 1
    fi

    echo ""
    echo "All automated checks passed!"
    echo ""
    echo "============================================"
    echo "MANUAL CHECKPOINT REQUIRED"
    echo "============================================"
    echo "Send a message to your Telegram bot and verify Rachel responds."
    echo "Container: ${CONTAINER_NAME}"
    echo "Logs: docker logs -f ${CONTAINER_NAME}"
    echo ""
    ```
