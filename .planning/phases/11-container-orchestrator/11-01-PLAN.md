    **Implementation:**

    ```typescript
    import { config, log } from "./config";
    import type {
      DockerContainerListItem,
      DockerContainerInspect,
      DockerCreateResponse,
      DockerCreateContainerBody,
      DockerNetworkInspect,
      DockerVolumeInspect,
    } from "./types";

    const API_BASE = `http://localhost/${config.dockerApiVersion}`;

    // --- Core fetch helper ---

    async function dockerFetch(
      path: string,
      options: RequestInit = {},
    ): Promise<Response> {
      const url = `${API_BASE}${path}`;
      const res = await fetch(url, {
        ...options,
        // @ts-expect-error -- Bun extension for Unix socket
        unix: config.dockerSocket,
      });
      return res;
    }

    async function dockerJson<T>(path: string, options: RequestInit = {}): Promise<T> {
      const res = await dockerFetch(path, options);
      if (!res.ok) {
        const body = await res.text();
        throw new DockerApiError(res.status, path, body);
      }
      return res.json() as Promise<T>;
    }

    export class DockerApiError extends Error {
      constructor(
        public status: number,
        public path: string,
        public body: string,
      ) {
        super(`Docker API error ${status} on ${path}: ${body}`);
        this.name = "DockerApiError";
      }
    }
    ```

    **Container operations:**

    ```typescript
    export const docker = {
      // --- Containers ---

      async listContainers(all = false): Promise<DockerContainerListItem[]> {
        return dockerJson(`/containers/json?all=${all}`);
      },

      async inspectContainer(nameOrId: string): Promise<DockerContainerInspect> {
        return dockerJson(`/containers/${encodeURIComponent(nameOrId)}/json`);
      },

      async createContainer(
        name: string,
        body: DockerCreateContainerBody,
      ): Promise<DockerCreateResponse> {
        return dockerJson(`/containers/create?name=${encodeURIComponent(name)}`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(body),
        });
      },

      async startContainer(nameOrId: string): Promise<void> {
        const res = await dockerFetch(
          `/containers/${encodeURIComponent(nameOrId)}/start`,
          { method: "POST" },
        );
        if (res.status === 204 || res.status === 304) return; // success or already running
        throw new DockerApiError(res.status, "start", await res.text());
      },

      async stopContainer(nameOrId: string, timeoutSec = 10): Promise<void> {
        const res = await dockerFetch(
          `/containers/${encodeURIComponent(nameOrId)}/stop?t=${timeoutSec}`,
          { method: "POST" },
        );
        if (res.status === 204 || res.status === 304) return; // stopped or already stopped
        if (res.status === 404) return; // container doesn't exist
        throw new DockerApiError(res.status, "stop", await res.text());
      },

      async restartContainer(nameOrId: string, timeoutSec = 10): Promise<void> {
        const res = await dockerFetch(
          `/containers/${encodeURIComponent(nameOrId)}/restart?t=${timeoutSec}`,
          { method: "POST" },
        );
        if (res.status === 204) return;
        throw new DockerApiError(res.status, "restart", await res.text());
      },

      async removeContainer(nameOrId: string, force = false): Promise<void> {
        const res = await dockerFetch(
          `/containers/${encodeURIComponent(nameOrId)}?force=${force}`,
          { method: "DELETE" },
        );
        if (res.status === 204) return;
        if (res.status === 404) return; // already removed
        throw new DockerApiError(res.status, "remove", await res.text());
      },

      // --- Volumes ---

      async createVolume(name: string): Promise<DockerVolumeInspect> {
        return dockerJson("/volumes/create", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ Name: name }),
        });
      },

      async removeVolume(name: string): Promise<void> {
        const res = await dockerFetch(
          `/volumes/${encodeURIComponent(name)}`,
          { method: "DELETE" },
        );
        if (res.status === 204 || res.status === 404) return;
        throw new DockerApiError(res.status, "removeVolume", await res.text());
      },

      async inspectVolume(name: string): Promise<DockerVolumeInspect | null> {
        try {
          return await dockerJson(`/volumes/${encodeURIComponent(name)}`);
        } catch (err) {
          if (err instanceof DockerApiError && err.status === 404) return null;
          throw err;
        }
      },

      // --- Networks ---

      async createNetwork(
        name: string,
        options?: { enableIcc?: boolean },
      ): Promise<{ Id: string }> {
        return dockerJson("/networks/create", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            Name: name,
            Driver: "bridge",
            Options: {
              "com.docker.network.bridge.enable_icc": String(options?.enableIcc ?? false),
            },
          }),
        });
      },

      async listNetworks(): Promise<DockerNetworkInspect[]> {
        return dockerJson("/networks");
      },

      // --- Utility ---

      async ping(): Promise<boolean> {
        try {
          const res = await dockerFetch("/_ping");
          return res.ok;
        } catch {
          return false;
        }
      },
    };
    ```

    Each method:
    - Uses `encodeURIComponent()` for container/volume names to handle special chars
    - Handles common status codes gracefully (304 = already in state, 404 = doesn't exist)
    - Throws `DockerApiError` with status, path, and body for debugging
    - Stop and remove are lenient (no-op if container doesn't exist)

    1. `provisionContainer(userId: string, env: UserContainerEnv): Promise<ContainerStatus>`
       - Idempotent: if container exists and is running with correct image, return status
       - If container exists but stopped, start it
       - If container exists but wrong image, stop + remove + recreate
       - If container doesn't exist: ensure volume → ensure network → create → start
       - Build container config matching run-container.sh: memory, CPU, caps, read-only, etc.
       - Set env vars: TELEGRAM_BOT_TOKEN, OWNER_TELEGRAM_USER_ID, ANTHROPIC_BASE_URL, ANTHROPIC_AUTH_TOKEN, GROQ_API_KEY, etc.
       - Container name: `rachel-user-{userId}`
       - Volume name: `rachel-user-{userId}-data`
       - Return container status after provisioning
       - Log all actions with userId

    2. `deprovisionContainer(userId: string, removeData: boolean): Promise<void>`
       - Stop the container (graceful 10s timeout)
       - Remove the container
       - If removeData=true, also remove the named volume (destructive!)
       - Log a warning when removing data
       - No-op if container doesn't exist

    3. `restartContainer(userId: string): Promise<ContainerStatus>`
       - Restart the container via Docker API
       - Return new status

    4. `getContainerStatus(userId: string): Promise<ContainerStatus | null>`
       - Inspect container, return typed status
       - Return null if container doesn't exist

    5. `listAllContainers(): Promise<ContainerStatus[]>`
       - List all containers with `rachel-user-` prefix
       - Return array of typed statuses (inspect each for health info)
