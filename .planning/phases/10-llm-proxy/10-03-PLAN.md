    **Part A: Add retry logic to `/home/rachel/rachel-cloud/src/proxy/forward.ts`**

    Add a `fetchWithRetry` helper function used by `forwardToZai()` for non-streaming requests only.

    ```typescript
    const MAX_RETRIES = 3;
    const RETRY_DELAYS = [1000, 2000, 4000]; // Exponential backoff: 1s, 2s, 4s

    async function fetchWithRetry(
      url: string,
      options: RequestInit,
      maxRetries: number = MAX_RETRIES,
    ): Promise<Response> {
      let lastError: Error | null = null;

      for (let attempt = 0; attempt <= maxRetries; attempt++) {
        try {
          const response = await fetch(url, options);

          // Don't retry client errors (4xx) -- those are the container's problem
          if (response.status < 500) {
            return response;
          }

          // 5xx -- retry if we have attempts left
          if (attempt < maxRetries) {
            log.warn("Upstream 5xx, retrying", {
              status: response.status,
              attempt: attempt + 1,
              maxRetries,
            });
            await Bun.sleep(RETRY_DELAYS[attempt] || 4000);
            continue;
          }

          // Final attempt failed with 5xx -- return the 5xx response as-is
          return response;
        } catch (error) {
          lastError = error as Error;

          if (attempt < maxRetries) {
            log.warn("Upstream network error, retrying", {
              error: String(error),
              attempt: attempt + 1,
              maxRetries,
            });
            await Bun.sleep(RETRY_DELAYS[attempt] || 4000);
            continue;
          }
        }
      }

      // All retries exhausted with network errors
      throw lastError || new Error("All retry attempts failed");
    }
    ```

    **Modify forwardToZai():**
    - For non-streaming requests (`stream` is false or absent in body): use `fetchWithRetry()` instead of plain `fetch()`
    - For streaming requests (`stream: true`): use plain `fetch()` -- do NOT retry because SSE streams cannot be replayed mid-stream
    - Add a timeout to the fetch signal: `AbortSignal.timeout(120_000)` (120 seconds) for both streaming and non-streaming requests. This prevents hung connections.

    **Also add a request_id** to each request for log correlation:
    - Generate with `crypto.randomUUID()` at the start of `forwardToZai()`
    - Include in all log statements within the function
    - Include in the UsageData output
    - Add `requestId` field to UsageData interface

    **Update error responses to be Anthropic-compatible:**
    When the proxy itself generates an error (not passing through from Z.ai), use this format:
    ```typescript
    function proxyErrorResponse(status: number, errorType: string, message: string): Response {
      return new Response(
        JSON.stringify({
          type: "error",
          error: { type: errorType, message },
        }),
        {
          status,
          headers: { "content-type": "application/json" },
        },
      );
    }
    ```

    Use this for:
    - 502 on network errors: `proxyErrorResponse(502, "proxy_error", "Failed to reach upstream API after retries")`
    - 504 on timeout: `proxyErrorResponse(504, "timeout_error", "Upstream request timed out")`

    Create a comprehensive health check endpoint.

    ```typescript
    import { db } from "./db";
    import { config, log } from "./config";

    export interface HealthStatus {
      status: "ok" | "degraded" | "error";
      uptime: number;
      version: string;
      checks: {
        database: { status: "ok" | "error"; latencyMs: number; error?: string };
        upstream: { status: "ok" | "error" | "unknown"; latencyMs?: number; error?: string };
      };
      stats?: {
        totalRequests: number;
        totalTokens: number;
        uniqueUsers: number;
        last24h: {
          requests: number;
          inputTokens: number;
          outputTokens: number;
        };
      };
    }

    export async function handleHealth(detailed: boolean = false): Promise<Response> {
      const health: HealthStatus = {
        status: "ok",
        uptime: process.uptime(),
        version: "1.0.0",
        checks: {
          database: { status: "ok", latencyMs: 0 },
          upstream: { status: "unknown" },
        },
      };

      // Check database
      try {
        const start = performance.now();
        db.prepare("SELECT 1").get();
        health.checks.database.latencyMs = Math.round(performance.now() - start);
      } catch (err) {
        health.checks.database.status = "error";
        health.checks.database.error = String(err);
        health.status = "error";
      }

      // Check upstream (lightweight HEAD request to Z.ai base URL)
      // Only check on detailed health requests to avoid hammering Z.ai
      if (detailed) {
        try {
          const start = performance.now();
          const resp = await fetch(config.upstreamBaseUrl, {
            method: "HEAD",
            signal: AbortSignal.timeout(5000),
          });
          health.checks.upstream.latencyMs = Math.round(performance.now() - start);
          health.checks.upstream.status = resp.ok || resp.status === 405 ? "ok" : "error";
        } catch (err) {
          health.checks.upstream.status = "error";
          health.checks.upstream.error = String(err);
          health.status = "degraded";
        }
      }

      // Include stats if detailed
      if (detailed) {
        try {
          const now = Math.floor(Date.now() / 1000);
          const dayAgo = now - 86400;

          const total = db.prepare(`
            SELECT COUNT(*) as requests, COALESCE(SUM(input_tokens + output_tokens), 0) as tokens
            FROM usage
          `).get() as { requests: number; tokens: number };

          const users = db.prepare(`
            SELECT COUNT(DISTINCT user_id) as count FROM usage
          `).get() as { count: number };

          const last24h = db.prepare(`
            SELECT
              COUNT(*) as requests,
              COALESCE(SUM(input_tokens), 0) as input_tokens,
              COALESCE(SUM(output_tokens), 0) as output_tokens
            FROM usage
            WHERE timestamp > ?
          `).get(dayAgo) as { requests: number; input_tokens: number; output_tokens: number };

          health.stats = {
            totalRequests: total.requests,
            totalTokens: total.tokens,
            uniqueUsers: users.count,
            last24h: {
              requests: last24h.requests,
              inputTokens: last24h.input_tokens,
              outputTokens: last24h.output_tokens,
            },
          };
        } catch (err) {
          log.error("Failed to gather health stats", { error: String(err) });
        }
      }

      const statusCode = health.status === "ok" ? 200 : health.status === "degraded" ? 200 : 503;

      return new Response(JSON.stringify(health, null, 2), {
        status: statusCode,
        headers: { "content-type": "application/json" },
      });
    }
    ```

    The `detailed` parameter is triggered by `GET /health?detailed=true`. The basic `/health` endpoint is fast (just DB ping, no upstream check).
