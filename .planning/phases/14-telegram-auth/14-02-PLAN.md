### `src/lib/auth/telegram-login.ts` — Hash Verification
```typescript
import { createHmac, createHash } from 'node:crypto';

interface TelegramLoginData {
  id: number;
  first_name: string;
  last_name?: string;
  username?: string;
  photo_url?: string;
  auth_date: number;
  hash: string;
}

/**
 * Verify Telegram Login Widget data using HMAC-SHA256.
 * @see https://core.telegram.org/widgets/login#checking-authorization
 */
export function verifyTelegramLogin(data: TelegramLoginData, botToken: string): boolean {
  const { hash, ...rest } = data;

  // 1. Sort fields alphabetically, format as key=value\n
  const checkString = Object.keys(rest)
    .sort()
    .map(k => `${k}=${rest[k as keyof typeof rest]}`)
    .join('\n');

  // 2. Secret key = SHA256(bot_token)
  const secretKey = createHash('sha256').update(botToken).digest();

  // 3. HMAC-SHA256(data_check_string, secret_key) must equal hash
  const hmac = createHmac('sha256', secretKey).update(checkString).digest('hex');

  if (hmac !== hash) return false;

  // 4. Check auth_date is recent (within 1 hour)
  const now = Math.floor(Date.now() / 1000);
  if (now - data.auth_date > 3600) return false;

  return true;
}
```

### `src/lib/auth/session.ts` — Lightweight Session Management
Replace Better Auth sessions with signed cookies:
```typescript
import { createHmac } from 'node:crypto';

const SESSION_SECRET = process.env.SESSION_SECRET || process.env.ENCRYPTION_KEY;
const SESSION_NAME = 'tg_session';
const SESSION_MAX_AGE = 7 * 24 * 60 * 60; // 7 days

interface SessionData {
  telegramId: number;
  firstName: string;
  username?: string;
  photoUrl?: string;
}

export function createSessionCookie(data: SessionData): string {
  const payload = Buffer.from(JSON.stringify(data)).toString('base64url');
  const signature = createHmac('sha256', SESSION_SECRET).update(payload).digest('base64url');
  return `${payload}.${signature}`;
}

export function verifySessionCookie(cookie: string): SessionData | null {
  const [payload, signature] = cookie.split('.');
  if (!payload || !signature) return null;

  const expected = createHmac('sha256', SESSION_SECRET).update(payload).digest('base64url');
  if (expected !== signature) return null;

  try {
    return JSON.parse(Buffer.from(payload, 'base64url').toString());
  } catch {
    return null;
  }
}
```

### `src/routes/api/auth/telegram/+server.ts` — Login Endpoint
```typescript
// POST /api/auth/telegram
// Receives Telegram Login Widget data, verifies, creates session
export async function POST({ request, cookies }) {
  const data = await request.json();

  if (!verifyTelegramLogin(data, process.env.TELEGRAM_BOT_TOKEN!)) {
    return json({ error: 'Invalid login' }, { status: 401 });
  }

  // Upsert user in DB
  await upsertUser(data);

  // Set session cookie
  const session = createSessionCookie({
    telegramId: data.id,
    firstName: data.first_name,
    username: data.username,
    photoUrl: data.photo_url,
  });

  cookies.set('tg_session', session, {
    path: '/',
    httpOnly: true,
    secure: true,
    sameSite: 'lax',
    maxAge: 7 * 24 * 60 * 60,
  });

  return json({ success: true, redirectTo: '/dashboard' });
}
```

    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(user),
  }).then(r => r.json()).then(data => {
    if (data.success) window.location.href = data.redirectTo;
  });
}

## `hooks.server.ts` Update
Replace Better Auth handler with session cookie verification:
```typescript
export const handle: Handle = async ({ event, resolve }) => {
  // Read session from cookie
  const sessionCookie = event.cookies.get('tg_session');
  event.locals.session = sessionCookie ? verifySessionCookie(sessionCookie) : null;

  // Admin guard
  if (event.url.pathname.startsWith('/admin')) {
    if (!event.locals.session ||
        event.locals.session.telegramId !== Number(process.env.ADMIN_TELEGRAM_ID)) {
      throw redirect(302, '/login');
    }
  }

  // App route guard
  if (event.url.pathname.startsWith('/dashboard')) {
    if (!event.locals.session) {
      throw redirect(302, '/login');
    }
  }

  return resolve(event);
};
```
